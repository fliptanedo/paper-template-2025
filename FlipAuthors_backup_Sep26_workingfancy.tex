%%   ENGINE   %%

% 1. Define institution database
\newcommand{\defineinst}[2]{%
  \pgfkeys{/inst data/#1/.initial={#2}}%
}
\newcommand{\getinst}[1]{\pgfkeysvalueof{/inst data/#1}}

% 2. Define author database
\newcommand{\defineauthor}[5]{%
  \pgfkeys{/author data/#1/name/.initial={#2}}%
  \pgfkeys{/author data/#1/email/.initial={#3}}%
  \pgfkeys{/author data/#1/orcid/.initial={#4}}%
  \pgfkeys{/author data/#1/instkeys/.initial={#5}}%
}

\newcommand{\getname}[1]{\pgfkeysvalueof{/author data/#1/name}}
\newcommand{\getemail}[1]{\pgfkeysvalueof{/author data/#1/email}}
\newcommand{\getorcid}[1]{\pgfkeysvalueof{/author data/#1/orcid}}
\newcommand{\getauthorinsts}[1]{\pgfkeysvalueof{/author data/#1/instkeys}}

\newcommand{\authororder}{yourname, tanedo, tunombre}

\newcounter{totalauthors}
\newcounter{authorindex}
\newcounter{instcount}
\newcounter{tmpcnt}
\newcounter{loopidx}

\gdef\orderedinstkeys{}

% Step 1: Parse authors and assign global map keys to unique institutions
\setcounter{instcount}{0}%
\foreach \auth in \authororder {%
  \edef\currentinsts{\getauthorinsts{\auth}}%
  \foreach \ikey in \currentinsts {%
    \ifcsname instnum@\ikey\endcsname\else
      \stepcounter{instcount}%
      \expandafter\xdef\csname instnum@\ikey\endcsname{\the\value{instcount}}%
      \ifx\orderedinstkeys\empty
        \xdef\orderedinstkeys{\ikey}%
      \else
        \xdef\orderedinstkeys{\orderedinstkeys,\ikey}%
      \fi
    \fi
  }%
}%

% Step 2: Fixed Superscript Builder Engine (Prints inline, bypassing loop scoping)
\newcommand{\printauthorsuperscript}[1]{%
  \setcounter{loopidx}{0}%
  \edef\myinsts{\getauthorinsts{#1}}%
  \foreach \ikey in \myinsts {%
    \stepcounter{loopidx}%
    \edef\instnum{\csname instnum@\ikey\endcsname}%
    \setcounter{tmpcnt}{\instnum}%
    \ifnum\value{loopidx}>1 , \fi
    \alph{tmpcnt}%
  }%
}

\newcommand{\countauthors}{%
  \setcounter{totalauthors}{0}%
  \foreach \k in \authororder {%
    \stepcounter{totalauthors}%
  }%
}


%%%%%%%%%%%%%%%%%%
%% RENDER BLOCK %%
%%%%%%%%%%%%%%%%%%

\begin{center}
  \countauthors
  
  % Print Author Names with Superscript Letters
  \setcounter{authorindex}{0}%
  \foreach \key in \authororder {%
    \stepcounter{authorindex}%
    \ifnum\value{authorindex}>1 %
      \ifnum\value{authorindex}=\value{totalauthors} and \else, \fi
    \fi
    \textbf{\getname{\key}}$^{\printauthorsuperscript{\key}}$%
  }%
  \par\vspace{1em}

  % Print Emails and ORCIDs
  \setcounter{authorindex}{0}%
  \foreach \key in \authororder {%
    \stepcounter{authorindex}%
    \ifnum\value{authorindex}>1 , \fi
    \texttt{\footnotesize \email{\getemail{\key}}}~\orcidlink{\getorcid{\key}}%
  }%
\end{center}

% Print Unique Affiliations cleanly (a, b, c...)
\begin{quotation}\noindent
  \footnotesize
  \setcounter{tmpcnt}{0}%
  \foreach \ikey in \orderedinstkeys {%
    \stepcounter{tmpcnt}%
    \ifnum\value{tmpcnt}>1 \\ \fi
    $^{\alph{tmpcnt}}$~%
    \parbox[t]{0.92\textwidth}{%
      \textit{\getinst{\ikey}}%
    }%
  }%
\end{quotation}
